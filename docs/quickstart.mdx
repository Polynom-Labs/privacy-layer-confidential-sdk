# Quickstart

```ts
import { isPreparedOperation } from '@arcane/privacy-sdk-core';
import { createStellarPrivacyClient } from '@arcane/privacy-sdk-stellar';

const clientOrRejected = await createStellarPrivacyClient({
  network,
  wallet,
  storage,
  assets: {
    sdkWasm,
    circuitWasm,
    provingKey,
  },
});

if (!isPreparedOperation(clientOrRejected)) {
  throw new Error(clientOrRejected.errors.map((error) => error.message).join('; '));
}

const operation = await clientOrRejected.deposit({
  from: await wallet.getAddress(),
  to: recipientAddress,
  asset: usdcAsset,
  amount: 100_0000000n,
  disclosure: {
    senderAddress: 'public',
    recipientAddress: 'private',
    assetAddress: 'public',
    amount: 'public',
  },
});

if (operation.status === 'prepared') {
  await operation.execute({ onEvent: console.log });
}
```

Use the Node entrypoint when asset files should be loaded from disk.
